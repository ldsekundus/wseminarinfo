public class Preisberechnung {
    public static void main(String[] args) {
        double grundpreis = 100.0;
        double versand = 5.0;
        double rabattFaktor = 0.10; // 10% Rabatt

        double endpreis = grundpreis - (grundpreis * rabattFaktor);
        endpreis = endpreis + versand;

        System.out.println("Endpreis: " + endpreis);
    }
}
public class Notenauswertung {
    public static void main(String[] args) {
        int punkte = 85;
        int note = 6;

        if (punkte >= 90) {
            note = 1;
        } else if (punkte >= 75) {
            note = 2;
        } else if (punkte >= 60) {
            note = 3;
        } else {
            note = 4;
        }

        System.out.println("Note: " + note);
    }
}
public class Lagerbestand {
    public static void main(String[] args) {
        int bestand = 20;
        int[] lieferungen = {5, 10, 2};

        for (int i = 0; i < lieferungen.length; i++) {
            bestand = bestand + lieferungen[i];
        }

        bestand = bestand - 7;
        System.out.println("Bestand: " + bestand);
    }
}
